cmake_minimum_required(VERSION 3.24)
project(WindowAppearanceConfigurator VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Windows x64 with MSVC is required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(appearance STATIC src/config.cpp src/renderer.cpp src/runtime.cpp src/wire.cpp)
target_link_libraries(appearance PUBLIC dwmapi gdiplus user32 gdi32 advapi32 shell32)
target_include_directories(appearance PUBLIC include third_party PRIVATE src)
target_compile_options(appearance PUBLIC /utf-8 /W4 /permissive- /EHsc /Brepro)
target_compile_definitions(appearance PUBLIC _WIN32_WINNT=0x0A00 UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
add_executable(companion src/main.cpp)
target_link_libraries(companion PRIVATE appearance)
target_link_options(companion PRIVATE /Brepro)
add_executable(window_appearance_tests tests/tests.cpp)
target_link_libraries(window_appearance_tests PRIVATE appearance)
enable_testing()
add_test(NAME appearance_contract COMMAND window_appearance_tests contract)
add_test(NAME appearance_windows COMMAND window_appearance_tests windows "${CMAKE_CURRENT_SOURCE_DIR}/evidence")
